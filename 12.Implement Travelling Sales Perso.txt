12.Implement Travelling Sales Person problem using Branch and Bound approach.
import itertools
def tsp(graph):
 n = len(graph)
 vertices = list(range(n))
 min_path = float('inf')
 best_order = []
 for perm in itertools.permutations(vertices[1:]):
 current_pathweight = 0
 k = 0
 for j in perm:
 current_pathweight += graph[k][j]
 k = j
 current_pathweight += graph[k][0]
 if current_pathweight < min_path:
 min_path = current_pathweight
 best_order = (0,) + perm + (0,)
 return min_path, best_order
graph = [
 [0, 10, 15, 20],
 [10, 0, 35, 25],
 [15, 35, 0, 30],
 [20, 25, 30, 0]
]
min_cost, path = tsp(graph)
print("Minimum cost:", min_cost)
print("Path:", path)
output:
Minimum cost: 80
Path: (0, 1, 3, 2, 0)
