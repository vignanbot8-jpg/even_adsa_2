4.Write a program for finding the bi-connected components in a given graph
from collections import defaultdict
class Graph:
 def __init__(self, vertices):
 self.V = vertices
 self.graph = defaultdict(list)
 self.time = 0
 def add_edge(self, u, v):
 self.graph[u].append(v)
 self.graph[v].append(u)
 def _bcc_util(self, u, parent, disc, low, st, bcc_list):
 disc[u] = low[u] = self.time
 self.time += 1
 children = 0
 for v in self.graph[u]:
 if disc[v] == -1:
 children += 1
 st.append((u, v))
 self._bcc_util(v, u, disc, low, st, bcc_list)
 low[u] = min(low[u], low[v])
 if (parent == -1 and children > 1) or (parent != -1 and low[v] >= disc[u]):
 bcc = []
 while st[-1] != (u, v):
 bcc.append(st.pop())
 bcc.append(st.pop())
 bcc_list.append(bcc)
 elif v != parent and disc[v] < disc[u]:
 low[u] = min(low[u], disc[v])
 st.append((u, v))
 def biconnected_components(self):
 disc = [-1] * self.V
 low = [-1] * self.V
 st = []
 bcc_list = []
 for i in range(self.V):
 if disc[i] == -1:
 self._bcc_util(i, -1, disc, low, st, bcc_list)
 if st:
 bcc = []
 while st:
 bcc.append(st.pop())
 bcc_list.append(bcc)
 return bcc_list
g = Graph(5)
edges = [(0,1),(1,2),(2,0),(1,3),(3,4)]
for u, v in edges:
 g.add_edge(u, v)
bccs = g.biconnected_components()
for i, comp in enumerate(bccs, 1):
 print(f"Biconnected Component {i}: {comp}")
output:
Biconnected Component 1: [(3, 4)]
Biconnected Component 2: [(1, 3)]
Biconnected Component 3: [(2, 0), (1, 2), (0, 1)]
